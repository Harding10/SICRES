Migrations SQL PostgreSQL

✓ 01_roles_permissions_migration.sql
  ├─ CREATE TABLE roles
  │   └─ id, name (UNIQUE), display_name, description
  │   └─ Insert: admin, school_director
  │
  ├─ CREATE TABLE permissions
  │   └─ id, name (UNIQUE), description
  │   └─ Insert: 28 permissions (users.*, schools.*, campaigns.*, etc)
  │
  ├─ CREATE TABLE role_permission (PIVOT)
  │   └─ role_id, permission_id (PK)
  │   └─ Assign all permissions to admin role
  │   └─ Assign specific permissions to school_director
  │
  └─ ALTER TABLE users
      └─ ADD role_id (FK to roles)

✓ 02_etablissements_migration.sql
  └─ CREATE TABLE etablissements
      ├─ id BIGSERIAL PK
      ├─ code (UNIQUE), name, type (public/privé)
      ├─ level, city, address
      ├─ director_id, director_name, director_phone, director_email
      ├─ phone, email, created_year
      ├─ latitude, longitude (geographic)
      ├─ status (active/closed/archived)
      ├─ created_at, updated_at, deleted_at (soft delete)
      └─ Indexes: type, city, level, director_id, status, deleted_at

✓ 03_campaigns_declarations_migration.sql
  ├─ CREATE TABLE campaigns
  │   ├─ id, title, description, reference_year
  │   ├─ opening_date, closing_date
  │   ├─ status (draft/active/closed)
  │   ├─ is_archived, archived_at
  │   ├─ created_by (FK to users)
  │   └─ created_at, updated_at
  │   └─ Indexes: status, reference_year, is_archived
  │
  ├─ CREATE TABLE declarations
  │   ├─ id, campaign_id (FK), etablissement_id (FK)
  │   ├─ status (draft/submitted/pending_review/validated/rejected)
  │   ├─ student_count, teacher_count, admin_staff_count
  │   ├─ classrooms_count, laboratories_count, libraries_count
  │   ├─ sports_facilities, priority_needs, observations
  │   ├─ submitted_at, validated_at, validated_by (FK)
  │   ├─ rejection_reason
  │   ├─ created_at, updated_at
  │   ├─ UNIQUE(campaign_id, etablissement_id)
  │   └─ Indexes: campaign_id, etablissement_id, status
  │
  └─ CREATE TABLE declaration_audits
      ├─ id, declaration_id (FK), action
      ├─ old_values (JSONB), new_values (JSONB)
      ├─ user_id (FK), ip_address, user_agent
      ├─ created_at
      └─ Indexes: declaration_id, user_id, created_at

✓ 04_documents_migration.sql
  └─ CREATE TABLE documents
      ├─ id, declaration_id (FK)
      ├─ type (decree/approval/plan/photos/reports)
      ├─ name, file_path, file_size, mime_type
      ├─ version, validation_status (pending/approved/rejected)
      ├─ uploaded_by (FK), validated_by (FK)
      ├─ rejection_reason
      ├─ uploaded_at, validated_at
      ├─ created_at, updated_at
      └─ Indexes: declaration_id, status, type, uploaded_by
